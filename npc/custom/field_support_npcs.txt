	set Zeny,Zeny-500;
	sc_start SC_BLESSING,600000,10;
	sc_start SC_INCREASEAGI,600000,10;
	sc_start SC_MAGNIFICAT,600000,5;
	sc_start SC_GLORIA,300000,5;
	sc_start SC_IMPOSITIO,300000,5;

	mes "[ผู้ช่วยภาคสนาม]";

	mes "ฉันให้เฉพาะบัฟที่ปลอดภัย ไม่รบกวนการใช้สกิลของอาชีพอื่นค่ะ";
	close;

	sc_end SC_STEELBODY;
	sc_end SC_PROVOKE;
	sc_end SC_SPIRIT;
	sc_end SC_EDP;
	sc_end SC_ENCPOISON;

	mes "[ผู้ช่วยภาคสนาม]";
