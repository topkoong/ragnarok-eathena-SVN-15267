// Purpose:
// - Repeatable item-proof hunting board.
// - Players collect monster drops and turn them in.
// - Rewards Zeny + account-bound Quest Coins.
//
// Currency:
// - #QuestCoin
// - Quest Coin Shop uses this same account variable.
//
// Why item-proof instead of kill counter?
// - Old eAthena versions can behave differently with kill events.
// - Item turn-in is safer, easier to maintain, and works reliably.
//============================================================

payon,189,100,4	script	HunterBoard#payon	837,{
	mes "[กระดานล่ามอนสเตอร์]";
	mes "ยินดีต้อนรับ นักล่าทุกท่าน";
	mes "เลือกภารกิจ เก็บวัตถุดิบจากมอนสเตอร์ แล้วนำมาส่งเพื่อรับรางวัล";
	next;

L_Main:
	mes "[กระดานล่ามอนสเตอร์]";
	mes "แต้มภารกิจของคุณ: ^0000FF" + #QuestCoin + "^000000 แต้ม";
	mes "เลือกประเภทภารกิจที่ต้องการค่ะ";
	next;
	menu "ภารกิจเริ่มต้น",L_Beginner,"ภารกิจระดับกลาง",L_Mid,"ภารกิจระดับสูง",L_Advanced,"คำแนะนำ",L_Info,"ยกเลิก",L_Close;


//============================================================
// Beginner missions
//============================================================

	mes "ภารกิจเริ่มต้น เหมาะสำหรับผู้เล่นใหม่หรือการฟาร์มเบา ๆ";
	next;
	menu "Jellopy x100",B_Jellopy,"Fluff x80",B_Fluff,"Sticky Webfoot x60",B_Webfoot,"Feather x60",B_Feather,"ย้อนกลับ",L_Main;

B_Jellopy:
	set .@item,909;
	set .@need,100;
	set .@coin,2;
	set .@zeny,5000;
	set .@name$,"Jellopy";
	goto L_TurnIn;

//============================================================
// Shared turn-in logic
//============================================================
L_TurnIn:
	mes "[กระดานล่ามอนสเตอร์]";
	mes "ภารกิจที่เลือก:";
	mes "^0000FF" + .@name$ + " x" + .@need + "^000000";
	mes "รางวัล:";
	mes "- Zeny: ^0000FF" + .@zeny + "^000000";
	mes "- Quest Coin: ^0000FF" + .@coin + "^000000 แต้ม";
	next;

	if (countitem(.@item) < .@need) {
		mes "[กระดานล่ามอนสเตอร์]";
		mes "คุณยังมีไอเทมไม่ครบค่ะ";
		mes "ต้องการ: ^FF0000" + .@name$ + " x" + .@need + "^000000";
		mes "มีอยู่: ^0000FF" + countitem(.@item) + "^000000";
		close;
	}

	menu "ส่งภารกิจ",L_Complete,"ยกเลิก",L_Close;

L_Complete:
	if (countitem(.@item) < .@need) {
		mes "[กระดานล่ามอนสเตอร์]";
		mes "ไอเทมไม่ครบค่ะ";
		close;
	}

	delitem .@item,.@need;
	set Zeny,Zeny+.@zeny;
	set #QuestCoin,#QuestCoin+.@coin;
	specialeffect2 248;

	mes "[กระดานล่ามอนสเตอร์]";
	mes "ส่งภารกิจเรียบร้อยค่ะ";
	mes "ได้รับ Zeny: ^0000FF" + .@zeny + "^000000";
	mes "ได้รับ Quest Coin: ^0000FF" + .@coin + "^000000 แต้ม";
	next;
	mes "[กระดานล่ามอนสเตอร์]";
	mes "แต้มภารกิจคงเหลือของคุณตอนนี้:";
	mes "^0000FF" + #QuestCoin + "^000000 แต้ม";
	close;

	next;
	mes "[กระดานล่ามอนสเตอร์]";
	mes "เมื่อส่งภารกิจ คุณจะได้รับ:";
	mes "- Zeny";
	mes "- Quest Coin";
	next;
	mes "[กระดานล่ามอนสเตอร์]";
	mes "Quest Coin สามารถนำไปแลกของรางวัลได้ที่ Prontera";
	mes "NPC: Quest Coin Shop";
	close;

L_Close:
	mes "[กระดานล่ามอนสเตอร์]";
	mes "ขอให้ล่าได้ของดีนะคะ";
	close;
