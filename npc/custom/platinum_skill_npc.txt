//============================================================
// Platinum Skill NPC
// eAthena SVN 15267 / Pre-Renewal High-Class style
//
// Purpose:
//   Grants classic platinum / quest skills.
//
// Notes:
//   - English/ASCII only. Safe to edit in VS Code.
//   - Uses numeric skill IDs for old eAthena compatibility.
//   - Some 2nd job quest skills may depend on your skill_db.txt.
//   - If a skill does not exist in your DB, it may simply fail at runtime.
//============================================================

prontera,166,82,4	script	Platinum Skill NPC	743,{

	//========================================================
	// Config
	//========================================================
	set .@Price, 0; // Change to 1000000 if you want to charge 1M zeny.

	mes "[Platinum Skill NPC]";
	mes "I can teach classic platinum and quest skills.";
	mes "These skills are useful for Pre-Renewal and High-Class gameplay.";
	next;

	if (.@Price > 0) {
		mes "[Platinum Skill NPC]";
		mes "Service fee: " + .@Price + " Zeny.";
		next;

		if (Zeny < .@Price) {
			mes "[Platinum Skill NPC]";
			mes "You do not have enough Zeny.";
			close;
		}
	}

	switch(select(
		"Learn First Job Platinum Skills",
		"Learn Second Job Quest Skills",
		"Learn All Available Skills",
		"Cancel"
	)) {
		case 1:
			if (.@Price > 0) set Zeny, Zeny - .@Price;
			callsub S_FirstJobSkills;
			mes "[Platinum Skill NPC]";
			mes "First job platinum skills have been granted.";
			close;

		case 2:
			if (.@Price > 0) set Zeny, Zeny - .@Price;
			callsub S_SecondJobSkills;
			mes "[Platinum Skill NPC]";
			mes "Second job quest skills have been granted.";
			close;

		case 3:
			if (.@Price > 0) set Zeny, Zeny - .@Price;
			callsub S_FirstJobSkills;
			callsub S_SecondJobSkills;
			mes "[Platinum Skill NPC]";
			mes "All available platinum and quest skills have been granted.";
			close;

		default:
			close;
	}


//============================================================
