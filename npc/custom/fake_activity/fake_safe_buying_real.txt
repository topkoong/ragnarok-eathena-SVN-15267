// Safe buying stores - prontera
prontera,200,218,4	script	B>YellowH#SB0001	748,{
	mes "[Buying Store]";
	mes "B> YellowHerb";
	mes "180 zeny each.";
	next;
	set @fp_have,countitem(508);
	if (@fp_have <= 0) {
		mes "You do not have this item.";
		close;
	}
	mes "You have " + @fp_have + " item(s).";
	mes "How many do you want to sell?";
	input @fp_amount;
	if (@fp_amount <= 0) close;
	if (@fp_amount > @fp_have) {
		mes "You do not have that many.";
		close;
	}
	set @fp_total,@fp_amount * 180;
	delitem 508,@fp_amount;
	set Zeny,Zeny+@fp_total;
	mes "Sold. You received " + @fp_total + " zeny.";
	set @fp_have,0;
	set @fp_amount,0;
	set @fp_total,0;
	close;

OnInit:
	initnpctimer;
	end;

OnTimer15000:
	npctalk "B> YellowHerb";
	end;

OnTimer35000:
	initnpctimer;
	end;
}
