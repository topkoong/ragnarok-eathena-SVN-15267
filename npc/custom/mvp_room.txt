// Purpose:
// - Controlled MVP challenge room.
// - One active MVP session at a time.

//============================================================

L_Confirm:
	mes "[ห้องท้าทาย MVP]";
	mes "คุณเลือก:";
	mes "^FF0000" + .@boss_name$ + "^000000";
	mes "ค่าเข้า: ^0000FF" + .@fee + "^000000 Zeny";
	mes "จำกัดเวลา: ^0000FF" + (.timeout_ms/60000) + "^000000 นาที";
	next;

	mes "[ห้องท้าทาย MVP]";
	mes "กติกาสำคัญ:";
	mes "- ห้องนี้มี MVP ได้ครั้งละ 1 ตัว";
	mes "- ถ้าหมดเวลา ระบบจะล้างห้อง";
	mes "- ไม่มีบัฟเต็ม ไม่มีฮีลฟรี และไม่มีซ่อมฟรีในห้อง";
	mes "- ถ้าตายหรือออกจากห้อง ถือว่าต้องรับผิดชอบเอง";
	next;

	if (Zeny < .@fee) {
		mes "[ห้องท้าทาย MVP]";
		mes "Zeny ของคุณไม่พอค่ะ";
		mes "ต้องใช้: ^FF0000" + .@fee + "^000000 Zeny";
		close;

L_Start:
	if (.busy == 1) {
		mes "[ห้องท้าทาย MVP]";
		mes "ขออภัยค่ะ มีคนเริ่มใช้ห้องไปก่อนแล้ว";
		close;
	}

	if (Zeny < .@fee) {
		mes "[ห้องท้าทาย MVP]";
		mes "Zeny ของคุณไม่พอค่ะ";
		close;
	}

	set Zeny,Zeny-.@fee;
	set #MVPRoomNextTime,gettimetick(2)+.@cooldown;

	set .busy,1;
	set .boss_id,.@boss_id;
	set .boss_name$,.@boss_name$;
	set .challenger$,strcharinfo(0);

	// Remove possible stale MVPs from previous failed sessions.
	killmonster .room_map$,"MVP Room#main::OnMVPDead";

	// Spawn selected MVP.
	monster .room_map$,50,50,.boss_name$,.boss_id,1,"MVP Room#main::OnMVPDead";

	// Start timeout timer.
	stopnpctimer;
	setnpctimer 0;
	initnpctimer;

	warp .room_map$,50,45;
	end;

	mes "3. ถ้าหมดเวลา MVP จะถูกล้างออก";
	mes "4. ไม่มีการคืนเงินถ้าแพ้ ตาย หรือออกจากห้อง";
	mes "5. ไม่ควรใส่ NPC บัฟเต็มไว้ในห้องนี้ เพราะจะทำให้ MVP Card ฟาร์มง่ายเกินไป";
	next;
	mes "[ห้องท้าทาย MVP]";
	mes "ถ้าต้องการปรับสมดุล ให้แก้ไขค่าที่ OnInit ด้านล่างของไฟล์:";
	mes "- ค่าเข้า";
	mes "- คูลดาวน์";
	mes "- เวลาจำกัด";
	mes "- แผนที่ห้อง";
	close;

L_Close:
	mes "[ห้องท้าทาย MVP]";
	mes "พร้อมเมื่อไหร่ค่อยกลับมาท้าทายใหม่นะคะ";
	close;


//============================================================
// MVP killed event
//============================================================
OnMVPDead:
	stopnpctimer;
	set .busy,0;
	announce "[MVP Room] " + .challenger$ + " has defeated " + .boss_name$ + "!",bc_all;
	set .boss_id,0;
	set .boss_name$,"";
	set .challenger$,"";
	end;


//============================================================
// Timeout cleanup
//============================================================
OnTimer900000:
	stopnpctimer;
	if (.busy == 1) {
		killmonster .room_map$,"MVP Room#main::OnMVPDead";
		announce "[MVP Room] The MVP challenge has ended because the time limit expired.",bc_all;
		set .busy,0;
		set .boss_id,0;
		set .boss_name$,"";
		set .challenger$,"";
	}
	end;


//============================================================
// Initial configuration
//============================================================
OnInit:
	// Change this if guild_vs1 does not exist in your server.
	set .room_map$,"guild_vs1";

	// Entry fees.
	set .easy_fee,1000000;
	set .normal_fee,2500000;
	set .hard_fee,5000000;

	// Cooldowns in seconds.
	set .easy_cd,3600;      // 1 hour
	set .normal_cd,7200;    // 2 hours
	set .hard_cd,10800;     // 3 hours

	// Timeout in milliseconds.
	set .timeout_ms,900000; // 15 minutes

	set .busy,0;
	set .boss_id,0;
	set .boss_name$,"";
	set .challenger$,"";
	end;
}
