//============================================================
// Reset Girl
//============================================================
prontera,159,208,4	script	Reset Girl	727,{

	mes "[Reset Girl]";
	mes "I can reset your status points, skill points, or both.";
	mes "Please choose carefully.";
	next;

	switch(select(
		"Reset Status",
		"Reset Skills",
		"Reset Both",
		"Cancel"
	)) {
		case 1:
			resetstatus;
			mes "[Reset Girl]";
			mes "Your status points have been reset.";
			close;

		case 2:
			resetskill;
			mes "[Reset Girl]";
			mes "Your skill points have been reset.";
			close;

		case 3:
			resetstatus;
			resetskill;
			mes "[Reset Girl]";
			mes "Your status and skill points have been reset.";
			close;

		default:
			close;
	}
}


//============================================================
// Identifier
// Old eAthena does not support identifyall as a script command.
// Use @identify through atcommand instead.
//============================================================
prontera,151,208,4	script	Identifier	731,{

	mes "[Identifier]";
	mes "I can identify all unidentified items in your inventory.";
	next;

	if (select("Identify all items", "Cancel") == 2) {
		close;
	}

	atcommand "@identify";

	mes "[Identifier]";
	mes "Done.";
	close;
}
