// 1) Censor Bar
// Official-style location: rachel,91,273
// Reward: 5175 Censor Bar / Anonymity Request
//============================================================
rachel,91,273,8	script	CensorMaker#2006	930,{
	if (checkweight(1201,1) == 0) {
		mes "[คนลึกลับ]";
		mes "ของในตัวคุณมากเกินไป";
		mes "ถ้าต้องการของสำหรับปกปิดตัวตน ต้องมีช่องว่างพอด้วยนะ...";
		close;
	}

	mes "[คนลึกลับ]";
	mes "ชู่ว... อย่าเสียงดัง";
	mes "ถ้าคุณอยากปกปิดตัวตน ฉันช่วยทำ Censor Bar ให้ได้";
	next;
	mes "[คนลึกลับ]";
	mes "วัตถุดิบที่ต้องใช้:";
	mes "- Black Dyestuffs x1";
	mes "- Slick Paper x100";
	mes "- Sticky Mucus x99";
	mes "- 100,000 Zeny";
	next;

	if (countitem(983) < 1 || countitem(7111) < 100 || countitem(938) < 99 || Zeny < 100000) {
		mes "[คนลึกลับ]";
		mes "วัตถุดิบหรือ Zeny ยังไม่ครบ";
		close;
	}

	menu "สร้าง Censor Bar",L_Make,"ยกเลิก",L_Close;

L_Make:
	if (countitem(983) < 1 || countitem(7111) < 100 || countitem(938) < 99 || Zeny < 100000) {
		mes "[คนลึกลับ]";
		mes "วัตถุดิบหรือ Zeny ยังไม่ครบ";
		close;
	}
	delitem 983,1;
	delitem 7111,100;
	delitem 938,99;
	set Zeny,Zeny-100000;
	getitem 5175,1;
	specialeffect2 248;
	mes "[คนลึกลับ]";
	mes "เสร็จแล้ว... รับ Censor Bar ไป";
	mes "อย่าให้ใครรู้ว่าได้มาจากฉันล่ะ";
	close;

L_Close:
	mes "[คนลึกลับ]";
	mes "ชู่ว... กลับไปเงียบ ๆ";
	close;
}
