//============================================================
// Buying logic
//============================================================
L_Buy:
	if (#QuestCoin < .@cost) {
		mes "[ร้านแลกแต้มภารกิจ]";
		mes "แต้มภารกิจของคุณไม่พอค่ะ";
		mes "ต้องใช้: ^FF0000" + .@cost + "^000000 แต้ม";
		mes "คุณมี: ^0000FF" + #QuestCoin + "^000000 แต้ม";
		close;
	}

	mes "[ร้านแลกแต้มภารกิจ]";
	mes "ยืนยันการแลกของรางวัลนี้หรือไม่?";
	mes "ใช้แต้ม: ^FF0000" + .@cost + "^000000 แต้ม";
	next;
	menu "ยืนยัน",L_ConfirmBuy,"ยกเลิก",L_Close;

L_ConfirmBuy:
	if (#QuestCoin < .@cost) {
		mes "[ร้านแลกแต้มภารกิจ]";
		mes "แต้มภารกิจของคุณไม่พอค่ะ";
		close;
	}
	set #QuestCoin,#QuestCoin-.@cost;
	getitem .@item,.@amount;
	specialeffect2 248;
	mes "[ร้านแลกแต้มภารกิจ]";
	mes "แลกของรางวัลเรียบร้อยแล้วค่ะ";
	mes "แต้มคงเหลือ: ^0000FF" + #QuestCoin + "^000000 แต้ม";
	close;

L_Close:

	mes "ขอบคุณที่ใช้บริการค่ะ";
