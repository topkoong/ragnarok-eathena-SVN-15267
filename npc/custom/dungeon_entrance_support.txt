//============================================================
// Dungeon Entrance Support NPCs
// File: npc/custom/dungeon_entrance_support.txt
// Server target: eAthena SVN 15267 / Pre-Renewal / High-Class
//
// Purpose:
// - Add lightweight support NPCs at dungeon entrances.
// - This is NOT a full Global Buffer.
// - These NPCs help players prepare before entering dungeons.
//
// Important:
// - field_support_npcs.txt already includes support for:
//      gef_dun00
//      moc_pryd01
//      glast_01
//   So this file intentionally avoids those locations.
//
// Services:
// - Heal HP/SP
// - Safe basic buffs
// - Identify items
// - Repair equipment using @repairall
// - Clear problematic buffs
// - Return to nearest town
// - Thai dungeon guidance
//
// Compatibility:
// - Uses atcommand "@identify" instead of identifyall;
// - Uses atcommand "@repairall" instead of repairall;
// - If your server does not support @repairall, remove only that line.
//
// Encoding note:
// - Because Thai dialogue is included, save this file as
//   Windows-874 / CP874 / TIS-620 when using it on old eAthena.
//============================================================

// ถ้ำ Payon
//============================================================
pay_dun00,21,183,4	script	Dungeon Support#paydun	837,{
	mes "[ผู้ช่วยหน้าดันเจี้ยน]";
	mes "ยินดีต้อนรับสู่ทางเข้า ^0000FFถ้ำ Payon^000000 ค่ะ";
	mes "ฉันช่วยฟื้นฟู เตรียมบัฟพื้นฐาน ซ่อมอุปกรณ์ และพากลับเมืองได้";
	next;

L_Main_paydun:
	mes "[ผู้ช่วยหน้าดันเจี้ยน]";
	mes "ต้องการใช้บริการใดคะ?";
	next;
	menu "ฟื้นฟู HP/SP",L_Heal_paydun,"รับบัฟพื้นฐาน",L_Buff_paydun,"ตรวจของ / ซ่อมอุปกรณ์",L_Service_paydun,"ล้างบัฟที่อาจทำให้ใช้สกิลไม่ได้",L_Clear_paydun,"คำแนะนำดันเจี้ยนนี้",L_Info_paydun,"กลับเมือง Payon",L_Return_paydun,"ยกเลิก",L_Close_paydun;

L_Heal_paydun:
	heal 999999,999999;
	specialeffect2 312;
	mes "[ผู้ช่วยหน้าดันเจี้ยน]";
	mes "ฟื้นฟู HP และ SP เรียบร้อยแล้วค่ะ";
	close;

L_Buff_paydun:
	if (Zeny < 500) {
		mes "[ผู้ช่วยหน้าดันเจี้ยน]";
		mes "ค่าบริการบัฟพื้นฐานคือ 500 Zeny ค่ะ";
		close;
	}
	set Zeny,Zeny-500;
	sc_start SC_BLESSING,600000,10;
	sc_start SC_INCREASEAGI,600000,10;
	sc_start SC_MAGNIFICAT,600000,5;
	sc_start SC_GLORIA,300000,5;
	sc_start SC_IMPOSITIO,300000,5;
	specialeffect2 338;
	mes "[ผู้ช่วยหน้าดันเจี้ยน]";
	mes "บัฟพื้นฐานเรียบร้อยแล้วค่ะ";
	mes "ฉันไม่ให้บัฟที่แรงเกินไป เพื่อให้การเล่นยังสมดุลนะคะ";
	close;

L_Service_paydun:
	mes "[ผู้ช่วยหน้าดันเจี้ยน]";
	mes "ฉันจะช่วยตรวจของที่ยังไม่ได้ Identify และซ่อมอุปกรณ์ให้ค่ะ";
	mes "ถ้าเซิร์ฟเวอร์ของคุณไม่มีคำสั่ง @repairall ให้ลบบรรทัดนั้นออกจากสคริปต์นี้ได้เลย";
	next;
	atcommand "@identify";
	atcommand "@repairall";
	specialeffect2 248;
	mes "[ผู้ช่วยหน้าดันเจี้ยน]";
	mes "ดำเนินการตรวจของและซ่อมอุปกรณ์เรียบร้อยแล้วค่ะ";
	close;

L_Clear_paydun:
	sc_end SC_STEELBODY;
	sc_end SC_PROVOKE;
	sc_end SC_SPIRIT;
	sc_end SC_EDP;
	sc_end SC_ENCPOISON;
	mes "[ผู้ช่วยหน้าดันเจี้ยน]";
	mes "ล้างบัฟที่อาจทำให้ใช้สกิลไม่ได้เรียบร้อยแล้วค่ะ";
	close;

L_Info_paydun:
	mes "[ผู้ช่วยหน้าดันเจี้ยน]";
	mes "ถ้ำ Payon เหมาะสำหรับผู้เล่นช่วงต้นถึงกลาง";
	next;
	mes "[ผู้ช่วยหน้าดันเจี้ยน]";
	mes "ควรพก Potion และระวังมอนสเตอร์ Undead ด้านใน";
	next;
	mes "[ผู้ช่วยหน้าดันเจี้ยน]";
	mes "ก่อนเข้าดันเจี้ยน แนะนำให้ตรวจสอบ:";
	mes "- น้ำหนักตัวละคร";
	mes "- Potion";
	mes "- Fly Wing / Butterfly Wing";
	mes "- อุปกรณ์ที่เสียหาย";
	close;

L_Return_paydun:
	warp "payon",160,98;
	end;

L_Close_paydun:
	mes "[ผู้ช่วยหน้าดันเจี้ยน]";
	mes "ขอให้ปลอดภัยและล่าได้ของดีนะคะ";
	close;
}
