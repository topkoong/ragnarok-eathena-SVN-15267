// Dungeon Buffers - Dungeon Maps Only
// eAthena SVN 15267
//
// Rule:
// - Towns use Healer + Global Buffer.
// - Dungeon maps use Dun Buff.
// - No Prontera town Dun Buff is loaded.
//============================================================

function	script	F_DunBuff	{
	mes "[Dungeon Buffer]";
	mes "I can recover your HP/SP and give basic buffs.";
	next;

	if (select("Buff me:Cancel") == 2) close;

	percentheal 100,100;
	sc_start SC_BLESSING,240000,10;
	sc_start SC_INCREASEAGI,240000,10;

	specialeffect2 248;

	mes "[Dungeon Buffer]";
	mes "Done. Good luck!";
	close;
}
