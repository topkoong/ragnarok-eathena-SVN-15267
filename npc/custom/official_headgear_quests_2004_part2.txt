		mes "จงสวมเกียรตินี้ด้วยความภาคภูมิใจ";
		close;
	}

	close;

L_GiveJellopy:
	if (countitem(909) < 1000) {
		mes "[Orc Warrior]";
		mes "Jellopy ไม่ครบ";
		close;
	}
	delitem 909,1000;
	set orcs_hero_hat,orcs_hero_hat+1;
	specialeffect2 248;
	mes "[Orc Warrior]";
	mes "ข้ารับ Jellopy แล้ว";
	mes "ความคืบหน้าใหม่: ^0000FF" + orcs_hero_hat + "/10^000000";
	close;

L_GiveVoucher:
	set .@amt,countitem(931);
	if (.@amt < 1) {
		mes "[Orc Warrior]";
		mes "เจ้าไม่มี Orcish Voucher";
		close;
	}
	delitem 931,.@amt;
	set orcs_hero_hat2,orcs_hero_hat2+.@amt;
	if (orcs_hero_hat2 >= 10000) {
		set orcs_hero_hat2,10000;
		set orcs_hero_hat,12;
		mes "[Orc Warrior]";
		mes "ครบ 10000 แล้ว";
		mes "กลับมาคุยกับข้าอีกครั้งเพื่อรับ Orc Helm";
		close;
	}
	mes "[Orc Warrior]";
	mes "ข้ารับ Orcish Voucher แล้ว";
	mes "ความคืบหน้า: ^0000FF" + orcs_hero_hat2 + "/10000^000000";
	close;

L_AskHero:
	if (countitem(968) < 100 || countitem(2299) < 1) {
		mes "[Orc Warrior]";
		mes "วัตถุดิบยังไม่ครบ";
		close;
	}
	mes "[Orc Warrior]";
	mes "นี่คือขั้นสุดท้าย";
	mes "หากส่งของแล้ว เจ้าจะได้รับ Helmet of Orc Hero";
	next;
	menu "ยืนยัน",L_MakeHeroHelm,"ยกเลิก",L_Close;

L_MakeHeroHelm:
	if (countitem(968) < 100 || countitem(2299) < 1) {
		mes "[Orc Warrior]";
		mes "วัตถุดิบยังไม่ครบ";
		close;
	}
	delitem 968,100;
	delitem 2299,1;
	getitem 5094,1;
	set orcs_hero_hat,14;
	specialeffect2 248;
	mes "[Orc Warrior]";
	mes "เจ้าคือมนุษย์ที่ได้รับการยอมรับจาก Orc Hero";
	mes "รับ Helmet of Orc Hero ไป";
	close;

L_Close:
	mes "[Orc Warrior]";
	mes "ออกไปได้แล้ว มนุษย์";
	close;
