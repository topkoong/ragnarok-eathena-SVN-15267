	end;

OnClock2000:
	if($NPC_GVG_GTime$!="ON")end;


	if(gettime(4)==2||gettime(4)==4||gettime(4)==6){


		AgitStart;
		announce "[ GuildWar ] : สงคราม Guild War ได้เริ่มต้นขึ้นแล้ว ณ บัดนี้ !!",0;
		callsub S_DisplayOwners;
	}
end;

OnClock2100:
	if($NPC_GVG_GTime$!="ON")end;


	if(gettime(4)==2||gettime(4)==4||gettime(4)==6){


		AgitEnd;
		announce "[ GuildWar ] : สงคราม Guild War ได้จบลงแล้ว !!",0;
		callfunc("NPC_RANK_OnAgitEnd");
		callsub S_DisplayOwners;
	}
end;

S_DisplayOwners:
	setarray .@maps$[0],"aldeg_cas01","aldeg_cas02","aldeg_cas03","aldeg_cas04","aldeg_cas05";
	setarray .@maps$[5],"gefg_cas01","gefg_cas02","gefg_cas03","gefg_cas04","gefg_cas05";
	setarray .@maps$[10],"payg_cas01","payg_cas02","payg_cas03","payg_cas04","payg_cas05";
	setarray .@maps$[15],"prtg_cas01","prtg_cas02","prtg_cas03","prtg_cas04","prtg_cas05";
	for( set .@i, 0; .@i <= 19; set .@i, .@i+1 ) {
		if (GetCastleData(.@maps$[.@i],1)) {
			Announce "บ้าน [" + GetCastleName(.@maps$[.@i]) + "] ถูกครอบครองโดยกิล [" + GetGuildName(GetCastleData(.@maps$[.@i],1)) + "]",bc_all|bc_woe;
		}	
	}
	end;
}
